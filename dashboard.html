async function loadReviewCounts() { const { data, error } = await supabase .from("google_review_counts") .select("*") .eq("restaurant", RESTAURANT) .order("captured_at", { ascending: false }); if (error) { console.error("Error cargando reseñas:", error); return; } const records = data || []; const today = dateKey(new Date()); const todayRecords = records.filter(record => dateKey(new Date(record.captured_at)) === today ); // Último total registrado hoy const latestToday = todayRecords[0] || null; // Primer total registrado hoy const firstToday = todayRecords.length ? todayRecords[todayRecords.length - 1] : null; // Sin registros de hoy, no se utiliza ningún total anterior const reviewCurrent = latestToday ? latestToday.review_count : "—"; const reviewBaseline = firstToday ? firstToday.review_count : "—"; const reviewsToday = firstToday && latestToday ? latestToday.review_count - firstToday.review_count : "—"; // Actualizar los valores del dashboard document.getElementById("reviewCurrent").textContent = reviewCurrent; document.getElementById("reviewBaseline").textContent = reviewBaseline; document.getElementById("reviewsToday").textContent = reviewsToday; const statusElement = document.getElementById("reviewStatus"); if (statusElement) { statusElement.textContent = todayRecords.length ? "Referencia del día" : "Sin registros de reseñas hoy"; } }
